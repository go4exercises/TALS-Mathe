\section*{3 · Musterlösung und Punkteraster}
\textbf{Allgemein:} Ganze Punkte je Zeile. Ein Fehler wird nur einmal abgezogen (Folgefehler: wer mit einem falschen
Zwischenergebnis richtig weiterrechnet, bekommt die Wegpunkte der folgenden Zeilen, aber keine (E)-Punkte). In der Spalte P:
\textbf{(A)}~= erkennen oder ablesen, zählt auch ohne Rechenweg · \textbf{(E)}~= Ergebnispunkt, Ergebnis richtig
\emph{und} Weg erkennbar · ohne Zeichen = die Zeile bewertet den Weg. Taschenrechner erlaubt; Längen und Flächen auf zwei
Dezimalen, Toleranz $\pm 0.01$ (oder durch gerundete Zwischenwerte erklärbar). \textbf{Einheiten:} Der erste Einheitenfehler im
ganzen Test kostet die (E)-Zeile dieses Ergebnisses, weitere nichts.
«Typische Fehler» nennt, welche Zeile 0 Punkte gibt, und die Punkte, die bleiben.
\textbf{Teilrichtige Zeilen:} Verlangt eine Zeile mehrere Angaben und stimmt nur ein Teil, gibt die Zeile 0 Punkte.

\begin{lpaufgabe}{G1}{Vierecke beschreiben}{4 P}
\begin{raster}
(a) Viereck & 1\A & Ein \textbf{Rechteck} (kein Quadrat).\\
(a) Begründung & 1 & Diagonalen, die sich halbieren: Parallelogramm. Gleich lang: Rechteck. Nicht senkrecht: kein Rhombus, also kein Quadrat. Nötig sind «gleich lang $\Rightarrow$ Rechteck» und «nicht senkrecht $\Rightarrow$ kein Quadrat».\\
(b) $\alpha$, $\beta$ & 1\E & Benachbarte Winkel ergänzen sich: $\alpha + \beta = 180^\circ$ und $\alpha - \beta = 40^\circ$, also $\alpha = 110^\circ$, $\beta = 70^\circ$. Beide nötig.\\
(b) $\gamma$, $\delta$ & 1 & Gegenüberliegende Winkel sind gleich: $\gamma = \alpha = 110^\circ$, $\delta = \beta = 70^\circ$. Folgepunkt: richtig aus den eigenen $\alpha$, $\beta$.\\
\end{raster}
\fehler{«Quadrat»: beide (a)-Zeilen 0 (die Diagonalen stehen ausdrücklich nicht senkrecht). «Rhombus» oder «gleichschenkliges Trapez»: beide (a)-Zeilen 0.
Mit der Winkelsumme $360^\circ$ statt $\alpha + \beta = 180^\circ$ gerechnet ($\beta = 160^\circ$, $\alpha = 200^\circ$ — kein Viereckswinkel): (b)-Zeile $\alpha$, $\beta$ 0; die $\gamma$-$\delta$-Zeile zählt, wenn $\gamma = \alpha$ und $\delta = \beta$ richtig zugeordnet sind $\to$ höchstens 3 von 4\,P.
$\gamma = 70^\circ$, $\delta = 110^\circ$ (gegenüber und benachbart vertauscht): $\gamma$-$\delta$-Zeile 0.}
\end{lpaufgabe}

\begin{lpaufgabe}{G2}{Parallelogramm}{4 P}
\begin{raster}
(a) Fläche & 1\E & $A = a \cdot h_a = 7.5 \cdot 4 = 30\,\text{cm}^2$\\
(a) Umfang & 1\E & $U = 2(a + b) = 2(7.5 + 5) = 25\,\text{cm}$\\
(b) Ansatz & 1 & Dieselbe Fläche mit $b$ als Grundseite: $A = b \cdot h_b$, also $h_b = \tfrac{A}{b}$. Folgepunkt: mit der eigenen Fläche.\\
(b) $h_b$ & 1\E & $h_b = \tfrac{30}{5} = 6\,\text{cm}$\\
\end{raster}
\fehler{$A = a \cdot b = 37.5$ (schräge Seite als Höhe): Flächenzeile 0; mit $h_b = \tfrac{37.5}{5} = 7.5$ zählt die Ansatzzeile als Folgepunkt, die $h_b$-Zeile nicht $\to$ 2 von 4\,P.
$h_b = \tfrac{2A}{b} = 12$ (Dreiecksformel): beide (b)-Zeilen 0.
$h_b = \tfrac{b \cdot h_a}{a} \approx 2.67$ (umgekehrtes Verhältnis): beide (b)-Zeilen 0.
$U = 2(7.5 + 4) = 23$ (Höhe statt Seite): Umfangzeile 0.}
\end{lpaufgabe}

\begin{lpaufgabe}{G3}{Fenster in Rhombusform}{4 P}
\begin{raster}
(a) Ansatz & 1 & $A = \tfrac{1}{2} \cdot e \cdot f = \tfrac{1}{2} \cdot 70 \cdot 40 = 1400\,\text{cm}^2$\\
(a) in m$^2$ & 1\E & $1400\,\text{cm}^2 = 0.14\,\text{m}^2$ (gleichwertig: $\tfrac{1}{2} \cdot 0.7 \cdot 0.4$)\\
(b) Seite & 1 & Rechtwinkliges Teildreieck mit den halben Diagonalen $35$ und $20$: $a = \sqrt{35^2 + 20^2} = \sqrt{1625} \approx 40.31\,\text{cm}$\\
(b) Umfang & 1\E & $U = 4a \approx 161.25\,\text{cm} \approx 1.61\,\text{m}$ (mit der gerundeten Seite $161.24\,\text{cm}$: zählt)\\
\end{raster}
\fehler{$A = e \cdot f = 2800\,\text{cm}^2 = 0.28\,\text{m}^2$ ($\tfrac{1}{2}$ vergessen): beide (a)-Zeilen 0.
$1400\,\text{cm}^2$ ohne Umrechnung in $\text{m}^2$: nur die Zeile «in m$^2$» 0 (eine fehlende Umrechnung, kein Einheitenfehler im Sinn der allgemeinen Regel) $\to$ 3 von 4\,P.
$a = \sqrt{70^2 + 40^2} \approx 80.62$ (ganze Diagonalen statt Hälften), $U \approx 322.49$: beide (b)-Zeilen 0.
$a = 35 + 20 = 55$ (Längen addiert): beide (b)-Zeilen 0.}
\end{lpaufgabe}

\begin{lpaufgabe}{G4}{Walmdach}{4 P}
\begin{raster}
(a) Mittellinie & 1\E & $m = \tfrac{1}{2}(12 + 7) = 9.5\,\text{m}$\\
(a) Ansatz Fläche & 1 & $A = m \cdot h$ (gleichwertig: $\tfrac{1}{2}(a + c) \cdot h$). Folgepunkt: mit der eigenen Mittellinie.\\
(a) Fläche & 1\E & $A = 9.5 \cdot 4.5 = 42.75\,\text{m}^2$\\
(b) Ziegel & 1 & $42.75 \cdot 15 = 641.25$, also mindestens $642$ Ziegel (aufgerundet). Folgepunkt: mit der eigenen Fläche, aufgerundet.\\
\end{raster}
\fehler{$A = (12 + 7) \cdot 4.5 = 85.5$ ($\tfrac{1}{2}$ vergessen): Ansatz- und Flächenzeile 0; die Ziegelzeile mit $85.5 \cdot 15 = 1282.5 \to 1283$ als Folgepunkt $\to$ 2 von 4\,P.
$641$ (abgerundet) oder $641.25$ (nicht gerundet): Ziegelzeile 0 — mit $641$ Ziegeln fehlt ein Stück Dach.
$m = 12 + 7 = 19$ oder $m = \tfrac{12 - 7}{2} = 2.5$: Mittellinienzeile 0; Ansatz zählt, wenn $A = m \cdot h$ dasteht.}
\end{lpaufgabe}

\begin{lpaufgabe}{G5}{Trapez rückwärts}{3 P}
\begin{raster}
Mittellinie & 1 & $m = \tfrac{A}{h} = \tfrac{60}{6} = 10\,\text{cm}$ (gleichwertig: $a + c = \tfrac{2A}{h} = 20\,\text{cm}$)\\
Umstellen & 1 & $m = \tfrac{1}{2}(a + c)$, also $c = 2m - a$ (gleichwertig: $c = (a + c) - a$). Folgepunkt: mit der eigenen Mittellinie.\\
$c$ & 1\E & $c = 20 - 13 = 7\,\text{cm}$\\
\end{raster}
\fehler{$c = \tfrac{2A}{h} = 20$ ($-\,a$ vergessen): Mittellinienzeile zählt ($a + c = 20$), Umstellen und $c$ 0 $\to$ 1 von 3\,P.
$c = \tfrac{A}{h} - a = -3$ (Mittellinie als $a + c$ genommen): Mittellinienzeile zählt, die anderen 0 $\to$ 1 von 3\,P; eine negative Länge ist kein Folgepunkt.}
\end{lpaufgabe}

\begin{lpaufgabe}{G6}{Gleichschenkliges Trapez}{4 P}
\begin{raster}
Überstand & 1 & Teildreieck aus Höhe, Schenkel und Überstand; Überstand $\tfrac{a - c}{2} = \tfrac{18 - 8}{2} = 5\,\text{cm}$\\
Höhe & 1\E & $h = \sqrt{13^2 - 5^2} = \sqrt{144} = 12\,\text{cm}$\\
Fläche & 1 & $A = \tfrac{1}{2}(18 + 8) \cdot 12 = 13 \cdot 12 = 156\,\text{cm}^2$. Folgepunkt: richtig mit der eigenen Höhe.\\
Umfang & 1\E & $U = 18 + 8 + 2 \cdot 13 = 52\,\text{cm}$\\
\end{raster}
\fehler{Ganzer Unterschied $18 - 8 = 10$ als Überstand ($h = \sqrt{69} \approx 8.31$, $A \approx 107.99$): Überstand- und Höhenzeile 0; Flächenzeile als Folgepunkt; Umfang zählt $\to$ 2 von 4\,P.
Schenkel als Höhe ($A = 13 \cdot 13 = 169$): Überstand-, Höhen- und Flächenzeile 0; Umfang zählt $\to$ 1 von 4\,P.
$h = \sqrt{13^2 + 5^2} \approx 13.93$ (plus statt minus): Höhenzeile 0, Überstand zählt, Fläche als Folgepunkt ($\approx 181.07$) $\to$ 3 von 4\,P.
$U = 18 + 8 + 13 = 39$ (nur ein Schenkel): Umfangzeile 0.}
\end{lpaufgabe}

\begin{lpaufgabe}{G7}{Eine fremde Lösung}{2 P}
\begin{raster}
Fehler & 1 & $d = a\sqrt{2}$ gilt nur im Quadrat (zwei gleiche Katheten). Im Rechteck sind die Katheten $6$ und $8$ verschieden. Nötig: «nur im Quadrat» oder eine gleichwertige Begründung.\\
Diagonale & 1\E & $d = \sqrt{6^2 + 8^2} = \sqrt{100} = 10\,\text{cm}$\\
\end{raster}
\fehler{$d = 8 \cdot \sqrt{2} \approx 11.31$ (Abkürzung mit der anderen Seite): beide Zeilen 0. $d = 6 + 8 = 14$: Diagonalenzeile 0.}
\end{lpaufgabe}
