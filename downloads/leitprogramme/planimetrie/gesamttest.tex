\begin{lpaufgabe}{G2}{Zwei Höhen, eine Fläche}{4 P}
\begin{minipage}[t]{0.55\linewidth}\vspace{0pt}
Im Dreieck $ABC$ ist $c = \overline{AB} = 10\,\text{cm}$, die zugehörige Höhe $h_c = 6\,\text{cm}$ und $a = \overline{BC} = 12\,\text{cm}$.
\begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=1pt]
\item Berechne den Flächeninhalt.
\item Berechne die Höhe $h_a$ zur Seite $a$.
\item Erkläre ohne Rechnung, warum $h_a$ kürzer ist als $h_c$.
\end{enumerate}
