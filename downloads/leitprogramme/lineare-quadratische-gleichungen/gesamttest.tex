\lpteil{Teil A · Umformen und Nullprodukt}{Kapitel 1 und 2 · 8 Punkte}

\begin{lpaufgabe}{G1}{Lineare Gleichung}{3 P}
Löse und mach die Probe: $\;2(3x - 4) - (x - 2) = 3x + 4$
\lpplatz{5}
\end{lpaufgabe}

\begin{lpaufgabe}{G2}{Lösungsfälle}{2 P}
Bestimme die Lösungsmenge und begründe mit dem, was beim Umformen übrig bleibt:
\begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=1pt]
\item $3(x + 2) - x = 2(x + 3)$
\item $5x - 2(x - 1) = 3x$
\end{enumerate}
\lpplatz{4}
\end{lpaufgabe}

\begin{lpaufgabe}{G3}{Ausklammern und Nullprodukt}{3 P}
Löse $4x^2 = 20x$ mit dem Satz vom Nullprodukt. Erkläre in einem Satz, was schiefgeht, wenn man die Gleichung zuerst durch $4x$ teilt.
\lpplatz{5}
\end{lpaufgabe}

\lpteil{Teil B · Quadratische Gleichungen lösen}{Kapitel 3 und 4 · 11 Punkte}

\begin{lpaufgabe}{G4}{Wurzelziehen und quadratische Ergänzung}{3 P}
\begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=1pt]
\item Löse $(x - 3)^2 = 25$.
\item Löse $x^2 + 6x - 16 = 0$ mit quadratischer Ergänzung.
\end{enumerate}
\lpplatz{6}
\end{lpaufgabe}

\begin{lpaufgabe}{G5}{Mitternachtsformel}{3 P}
Löse $2x^2 - 5x + 2 = 0$ mit der Mitternachtsformel. Gib auch die Diskriminante $D$ an.
\lpplatz{6}
\end{lpaufgabe}

\begin{lpaufgabe}{G6}{Das passende Verfahren}{5 P}
Bring jede Gleichung zuerst auf null, nenne das Verfahren, das du wählst, und löse:
\begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=1pt]
\item $9x^2 - 4 = 0$
\item $x^2 + 2x - 24 = 0$
\item $(x + 2)(x - 2) = x(x - 1)$
\end{enumerate}
\lpplatz{7}
\end{lpaufgabe}

\lpteil{Teil C · Parameterdiskussion}{Kapitel 5 · 6 Punkte}

\begin{lpaufgabe}{G7}{Lineare Gleichung mit Parameter}{3 P}
Löse $k^2 x - 1 = x + k$ nach $x$ auf ($k \in \mathbb{R}$). Unterscheide alle Fälle und gib jeweils die Lösungsmenge an.
\lpplatz{9}
\end{lpaufgabe}

\begin{lpaufgabe}{G8}{Quadratische Gleichung mit Parameter}{3 P}
Für welche $k$ hat $x^2 + kx + 9 = 0$ zwei, eine oder keine Lösung? Gib im Fall «eine Lösung» die Lösung an.
\lpplatz{9}
\end{lpaufgabe}
