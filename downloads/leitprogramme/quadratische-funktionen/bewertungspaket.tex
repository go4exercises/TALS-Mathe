\section*{3 · Musterlösung und Punkteraster}
\textbf{Allgemein:} Ganze Punkte je Zeile. Ein Fehler wird nur einmal abgezogen (Folgefehler). Gleichwertige Wege und
Schreibweisen zählen voll (Bruch oder Dezimalzahl, Faktoren in anderer Reihenfolge). In der Spalte P: \textbf{(E)}~= Ergebnispunkt,
\textbf{(A)}~= nur ablesen: Diese Punkte gibt es auch ohne Weg, alle anderen nur mit nachvollziehbarem Weg.
Teil A und C sind ohne Taschenrechner gestellt. «Typische Fehler» nennt, welche Zeile 0 Punkte gibt, und die Punkte, die bleiben.

\begin{lpaufgabe}{G1}{$f(x) = -(x-2)^2 + 9$}{4 P}
\begin{raster}
Scheitelpunkt & 1\A & $S(2 \mid 9)$\\
Öffnung & 1\A & nach unten, da $a = -1 < 0$\\
Nullstellen mit Weg & 1 & $(x-2)^2 = 9 \Rightarrow x - 2 = \pm 3 \Rightarrow x_1 = -1,\ x_2 = 5$ (beide nötig)\\
Produktform & 1\E & $f(x) = -(x+1)(x-5)$ (gleichwertig $(x+1)(5-x)$)\\
\end{raster}
\fehler{$S(-2 \mid 9)$ (Vorzeichen in der Klammer): Scheitelpunkt 0, Nullstellen $1$ und $-5$ und $-(x-1)(x+5)$ folgerichtig $\to$ 3 von 4\,P.
Nur eine Nullstelle ($\pm$ vergessen): Nullstellen 0, Produktform 0 (sie braucht beide Nullstellen) $\to$ 2 von 4\,P.
$(x+1)(x-5)$ ohne Minus: Produktform 0 $\to$ 3 von 4\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G2}{$f(x) = 2x^2 - 12x + 10$}{3 P}
\begin{raster}
Ansatz mit Weg & 1 & $2(x^2 - 6x + 5) = 2\big((x-3)^2 - 9 + 5\big)$ — oder $x_s = -\frac{b}{2a} = 3$, $y_s = f(3) = -8$\\
Scheitelform und Scheitel & 1\E & $f(x) = 2(x-3)^2 - 8$, $S(3 \mid -8)$\\
Schnittpunkt $y$-Achse & 1\A & $(0 \mid 10)$, denn $f(0) = c = 10$\\
\end{raster}
\fehler{$2(x-3)^2 - 4$ (die $-4$ nicht mit 2 multipliziert): Scheitelform 0 $\to$ 2 von 3\,P.
Scheitel über $-\frac{b}{2a}$ richtig, aber keine Scheitelform hingeschrieben: Scheitelform und Scheitel 0 $\to$ 2 von 3\,P.
$(0 \mid 10)$ fehlt: $y$-Achse 0 $\to$ 2 von 3\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G3}{Graph $\to$ Scheitelform}{2 P}
\begin{raster}
Scheitel abgelesen & 1\A & $S(-1 \mid -2)$\\
Streckfaktor und Gleichung & 1\E & von $S$ ein Schritt nach rechts $\to$ 1 hinauf, also $a = 1$: $f(x) = (x+1)^2 - 2$\\
\end{raster}
\fehler{$(x-1)^2 - 2$ (Vorzeichen in der Klammer): Gleichung 0 $\to$ 1 von 2\,P.
Verlangt ist die Scheitelform: Die gleichwertige Grundform $x^2 + 2x - 1$ gibt den Punkt «Gleichung»; den Punkt «Scheitel» nur, wenn $S(-1 \mid -2)$ angegeben ist.}
\end{lpaufgabe}

\begin{lpaufgabe}{G4}{$f(x) = 2x^2 - 4x - 30$}{3 P}
\begin{raster}
Anzahl mit $D$ begründet & 1 & $D = (-4)^2 - 4 \cdot 2 \cdot (-30) = 16 + 240 = 256 > 0$ $\Rightarrow$ zwei Nullstellen\\
Nullstellen mit Weg & 1 & $x_{1,2} = \frac{4 \pm 16}{4}$: $x_1 = -3,\ x_2 = 5$ — oder $2(x^2 - 2x - 15)$, zwei Zahlen mit Produkt $-15$ und Summe $2$\\
Produktform & 1\E & $f(x) = 2(x+3)(x-5)$\\
\end{raster}
\fehler{Anzahl ohne $D$ (nur aus den gefundenen Nullstellen): $D$-Zeile 0 $\to$ 2 von 3\,P.
$D = 16 - 240 = -224$ (Vorzeichen von $c$), daraus «keine Nullstellen»: $D$-Zeile und Nullstellen 0; die folgerichtige Aussage «keine Produktform» gibt den Punkt Produktform $\to$ 1 von 3\,P.
Nenner $2$ statt $2a$ ($x = 10$, $x = -6$): Nullstellen 0, $2(x-10)(x+6)$ folgerichtig $\to$ 2 von 3\,P.
$(x+3)(x-5)$ ohne den Faktor 2 oder $2(x-3)(x+5)$ (Vorzeichen in der Klammer): Produktform 0 $\to$ 2 von 3\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G5}{Scheitel $S(3 \mid -2)$, Punkt $P(5 \mid 6)$}{3 P}
\begin{raster}
Ansatz Scheitelform & 1 & $f(x) = a(x-3)^2 - 2$\\
$a$ bestimmt & 1 & $6 = a \cdot 4 - 2 \Rightarrow a = 2$\\
Gleichung & 1\E & $f(x) = 2(x-3)^2 - 2$ (gleichwertig $2x^2 - 12x + 16$)\\
\end{raster}
\fehler{Ansatz $a(x+3)^2 - 2$: Ansatz 0, $a = \tfrac18$ und $f(x) = \tfrac18(x+3)^2 - 2$ folgerichtig $\to$ 2 von 3\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G6}{Wasserstrahl}{3 P}
\begin{raster}
Ansatz & 1 & Nullstellen 0 und 8: $h(x) = a\,x(x-8)$ — oder Scheitel in der Mitte, $S(4 \mid 4)$: $h(x) = a(x-4)^2 + 4$\\
$a$ bestimmt & 1 & mit $S(4 \mid 4)$: $4 = a \cdot 4 \cdot (-4) \Rightarrow a = -0.25$ — oder mit $(0 \mid 0)$: $0 = a \cdot 16 + 4 \Rightarrow a = -0.25$\\
Gleichung & 1\E & $h(x) = -0.25\,x(x-8)$ bzw.\ $h(x) = -0.25(x-4)^2 + 4$, ausmultipliziert $-0.25x^2 + 2x$\\
\end{raster}
\fehler{$a = 0.25 > 0$ (Vorzeichenfehler, nach oben geöffnet, Widerspruch zum Sachverhalt): $a$-Zeile 0, Gleichung folgerichtig $\to$ 2 von 3\,P.
Scheitel bei $(8 \mid 4)$ statt in der Mitte: Ansatz 0, $a = -\tfrac{1}{16}$ und $-\tfrac{1}{16}(x-8)^2 + 4$ folgerichtig $\to$ 2 von 3\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G7}{$h(t) = -5t^2 + 20t + 1.5$}{3 P}
\begin{raster}
Zeitpunkt mit Weg & 1 & $t = -\frac{b}{2a} = -\frac{20}{2 \cdot (-5)} = 2$ — oder quadratisch ergänzen\\
Höhe & 1 & $h(2) = -20 + 40 + 1.5 = 21.5$\\
Antwort mit Einheiten & 1\E & Nach 2\,s ist der Ball am höchsten, 21.5\,m hoch ($t = 2$ liegt in der Flugzeit, $a = -5 < 0$: Maximum).\\
\end{raster}
\fehler{$t = -2$ (Minus in $-\frac{b}{2a}$): Zeitpunkt 0, $h(-2) = -58.5$ folgerichtig $\to$ 2 von 3\,P.
Antwort ohne Einheiten oder ohne Höhe: Antwort 0 $\to$ 2 von 3\,P.}
\end{lpaufgabe}

\begin{lpaufgabe}{G8}{Quadrat im Quadrat, Seite 8\,cm}{4 P}
\begin{raster}
Zielfunktion & 1 & Vom grossen Quadrat vier Dreiecke mit den Katheten $x$ und $8-x$ abziehen: $A(x) = 64 - 4 \cdot \frac12\,x(8-x)$ — oder Pythagoras: $A(x) = x^2 + (8-x)^2$\\
Grundform & 1 & $A(x) = 2x^2 - 16x + 64$\\
Stelle des Minimums mit Weg & 1 & $a = 2 > 0$, Minimum im Scheitel: $x = -\frac{b}{2a} = \frac{16}{4} = 4$ — oder quadratisch ergänzen: $2(x-4)^2 + 32$\\
Antwort mit Einheiten & 1\E & $x = 4$ liegt im zulässigen Bereich $0 \le x \le 8$. Bei $x = 4$\,cm (Punkte in den Seitenmitten) ist das innere Quadrat am kleinsten: $A(4) = 32$\,cm$^2$.\\
\end{raster}
\fehler{Dreiecke ohne $\frac12$, $A(x) = 4x^2 - 32x + 64$: Zielfunktion 0, $x = 4$ folgerichtig; $A(4) = 0$ ist unmöglich und gibt die Antwort nicht $\to$ 2 von 4\,P.
Nullstellen gesucht und keine gefunden ($D = 256 - 512 < 0$), darum keine Mitte: Stelle 0 $\to$ höchstens 3 von 4\,P.
$x = -4$ (Minus in $-\frac{b}{2a}$): Stelle 0; $x$ muss zwischen 0 und 8 liegen, die Antwort zählt nicht $\to$ 2 von 4\,P.}
\end{lpaufgabe}
