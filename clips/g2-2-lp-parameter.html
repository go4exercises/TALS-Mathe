\(kx + 6 = 2x + 3k\)
\(\fb{\mid -2x - 6}\)
\((k - 2) \cdot x = 3k - 6\)
\((k - 2) \cdot x = 3 \cdot (k - 2)\)
\((k - 2) \cdot x = 3 \cdot (k - 2)\)
\(k \neq 2: \quad x = \fc{3}\)
\(k = 2: \quad 0 = 0, \;\; \mathbb{L} = \mathbb{R}\)
\(\fb{k = 5}\)
