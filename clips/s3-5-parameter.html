\(y = \fa{a} \cdot \sin\bigl(\fb{b}(x - \fd{u})\bigr) + \fc{v}\)

\(\fc{v} \quad \longrightarrow \quad \text{Mittellinie } y = \fc{v}\)
\(\text{Hochpunkt } \fc{v} + \fa{a} \qquad \text{Tiefpunkt } \fc{v} - \fa{a}\)
